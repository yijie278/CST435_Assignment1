cmake_minimum_required(VERSION 3.16)
project(ParallelKMeans LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP)
add_executable(ParallelKMeans src/main.cpp)
if(OpenMP_CXX_FOUND)
 target_link_libraries(ParallelKMeans PRIVATE OpenMP::OpenMP_CXX)
else()
 message(WARNING "OpenMP not found: parallel measurements will not be valid until OpenMP is enabled.")
endif()
